cmake_minimum_required(VERSION 3.20)
project(FunkePathfindingTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(PathfindingCore STATIC ../src/Pathfinding.cpp)
target_include_directories(PathfindingCore PUBLIC ../include)
add_library(PathfindingNative SHARED ../src/Pathfinding.cpp ../src/PInvoke.cpp ../src/PackageEntry.cpp)
target_include_directories(PathfindingNative PRIVATE ../include)
set_target_properties(PathfindingNative PROPERTIES OUTPUT_NAME "Pkg.Funke.Pathfinding.Native")
add_executable(PathfindingTests PathfindingTests.cpp)
target_link_libraries(PathfindingTests PRIVATE PathfindingCore)
find_package(Threads REQUIRED)
target_link_libraries(PathfindingTests PRIVATE Threads::Threads)
enable_testing()
add_test(NAME Pathfinding COMMAND PathfindingTests)
add_executable(PathfindingBenchmark Benchmark.cpp)
target_link_libraries(PathfindingBenchmark PRIVATE PathfindingCore)
if(MSVC)
    foreach(target PathfindingCore PathfindingNative PathfindingTests PathfindingBenchmark)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    endforeach()
else()
    foreach(target PathfindingCore PathfindingNative PathfindingTests PathfindingBenchmark)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endforeach()
endif()
